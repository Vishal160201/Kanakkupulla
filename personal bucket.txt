SCHEMA — prisma/schema.prisma

model PersonalBucket {
  id                  String   @id @default(cuid())
  title               String
  type                String   // SAVINGS / DEBT / EMI
  targetAmount        Float
  savedAmount         Float    @default(0)
  monthlyContribution Float    @default(0)
  linkedCardId        String?
  dueDate             DateTime?
  notes               String?
  color               String   @default("#6366f1")
  icon                String   @default("Wallet")
  isActive            Boolean  @default(true)
  createdAt           DateTime @default(now())
  updatedAt           DateTime @updatedAt
  payments            PersonalBucketPayment[]
  @@index([linkedCardId])
  @@index([type])
}

model PersonalBucketPayment {
  id       String @id @default(cuid())
  bucketId String
  amount   Float
  month    Int    // 1-12
  year     Int
  status   String @default("PENDING") // PAID / NIL / PENDING
  notes    String?
  createdAt DateTime @default(now())
  bucket   PersonalBucket @relation(fields: [bucketId], references: [id])
  @@index([bucketId])
  @@unique([bucketId, month, year])
}

Run: npx prisma db push && npx prisma generate

API ROUTES — all under /api/personal/buckets/*
Every handler: if session.user.email !== 'nithyavishalr@gmail.com' return 403.

GET    /api/personal/buckets              — list all active buckets with payments[]
POST   /api/personal/buckets              — create bucket
PATCH  /api/personal/buckets/[id]         — edit bucket fields
DELETE /api/personal/buckets/[id]         — soft-delete (isActive: false)

GET    /api/personal/buckets/[id]/payments        — list payments for bucket (all months/years)
POST   /api/personal/buckets/[id]/payments        — create/upsert payment for month+year
PATCH  /api/personal/buckets/[id]/payments/[pid]  — update amount/status/notes
DELETE /api/personal/buckets/[id]/payments/[pid]  — delete payment entry

POST   /api/personal/buckets/[id]/contribute — add contribution:
  body: { amount, month, year, notes? }
  1. upsert PersonalBucketPayment { bucketId, month, year, amount, status:'PAID' }
  2. update PersonalBucket.savedAmount += amount
  3. if linkedCardId: create PersonalExpense { type:'DEBIT', category:'TRANSFER', amount, paymentSource: linkedCardId, title: 'Transfer to [bucket.title]', date: first day of that month }
  return updated bucket

GET /api/personal/buckets/summary — return:
  { totalAllocated: sum(savedAmount), totalTarget: sum(targetAmount), byType: { SAVINGS, DEBT, EMI }, freeBalance: linkedCard.manualBalance - totalAllocated }
